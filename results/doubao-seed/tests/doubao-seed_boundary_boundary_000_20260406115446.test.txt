from dataclasses import dataclass
from typing import TypedDict, Union, Literal, Optional, Sequence
import pytest
from . import (
    strict,
    validate_typed_dict,
    validated_field,
    StrictDataclassClassValidationError,
    StrictDataclassDefinitionError,
    StrictDataclassFieldValidationError,
)

def test_strict_decorator_non_dataclass_raises_definition_error():
    with pytest.raises(StrictDataclassDefinitionError, match="must be a dataclass before applying @strict"):
        @strict
        class NonDataclass:
            pass

def test_basic_strict_dataclass_accepts_correct_types():
    @dataclass
    @strict
    class TestDC:
        int_field: int
        str_field: str
        bool_field: bool

    obj = TestDC(int_field=42, str_field="test", bool_field=True)
    assert obj.int_field == 42
    assert obj.str_field == "test"
    assert obj.bool_field is True

def test_basic_strict_dataclass_rejects_invalid_types():
    @dataclass
    @strict
    class TestDC:
        int_field: int

    with pytest.raises(StrictDataclassFieldValidationError, match="field='int_field'"):
        TestDC(int_field="not an integer")

def test_custom_field_validator_works_correctly():
    def positive_validator(value: int) -> None:
        if value <= 0:
            raise ValueError("value must be positive")

    @dataclass
    @strict
    class TestDC:
        num: int = validated_field(positive_validator)

    obj = TestDC(num=10)
    assert obj.num == 10
    with pytest.raises(StrictDataclassFieldValidationError, match="must be positive"):
        TestDC(num=-5)

def test_invalid_custom_validator_raises_definition_error():
    invalid_validator = "not a callable"
    with pytest.raises(StrictDataclassDefinitionError, match="Invalid validator for field"):
        @dataclass
        @strict
        class TestDC:
            num: int = validated_field(invalid_validator)

def test_class_validator_runs_on_init():
    @dataclass
    @strict
    class TestDC:
        a: int
        b: int

        def validate_a_greater_than_b(self):
            if self.a <= self.b:
                raise ValueError("a must be larger than b")

    valid_obj = TestDC(a=10, b=5)
    assert valid_obj.a == 10
    with pytest.raises(StrictDataclassClassValidationError, match="validator='validate_a_greater_than_b'"):
        TestDC(a=3, b=5)

def test_existing_validate_method_raises_definition_error():
    with pytest.raises(StrictDataclassDefinitionError, match="already implements a method called 'validate'"):
        @dataclass
        @strict
        class TestDC:
            field: int
            def validate(self):
                pass

def test_accept_kwargs_allows_extra_fields():
    @dataclass
    @strict(accept_kwargs=True)
    class TestDC:
        required: int

    obj = TestDC(required=1, extra_str="test", extra_int=123)
    assert obj.required == 1
    assert obj.extra_str == "test"
    assert obj.extra_int == 123

def test_accept_kwargs_rejects_positional_args():
    @dataclass
    @strict(accept_kwargs=True)
    class TestDC:
        field: int

    with pytest.raises(ValueError, match="accepts only keyword arguments"):
        TestDC(1)

def test_accept_kwargs_raises_for_missing_required_fields():
    @dataclass
    @strict(accept_kwargs=True)
    class TestDC:
        required1: int
        required2: str

    with pytest.raises(TypeError, match="Missing required field - 'required2'"):
        TestDC(required1=1)

def test_list_type_validation():
    @dataclass
    @strict
    class TestDC:
        field: list[int]

    TestDC(field=[1, 2, 3])
    with pytest.raises(StrictDataclassFieldValidationError):
        TestDC(field=[1, "2", 3])
    with pytest.raises(StrictDataclassFieldValidationError):
        TestDC(field="not a list")

def test_dict_type_validation():
    @dataclass
    @strict
    class TestDC:
        field: dict[str, float]

    TestDC(field={"a": 1.1, "b": 2.2})
    with pytest.raises(StrictDataclassFieldValidationError):
        TestDC(field={"a": "1.1", "b": 2.2})
    with pytest.raises(StrictDataclassFieldValidationError):
        TestDC(field={1: 2.2})

def test_tuple_type_validation():
    @dataclass
    @strict
    class TestDC:
        fixed_tuple: tuple[int, str]
        var_tuple: tuple[bool, ...]

    TestDC(fixed_tuple=(1, "test"), var_tuple=(True, False, True))
    with pytest.raises(StrictDataclassFieldValidationError):
        TestDC(fixed_tuple=(1, 2), var_tuple=(True, "false"))
    with pytest.raises(StrictDataclassFieldValidationError):
        TestDC(fixed_tuple=(1,), var_tuple=(True, False))

def test_validate_typed_dict_accepts_valid_data():
    class User(TypedDict):
        id: int
        name: str
        is_active: bool

    validate_typed_dict(User, {"id": 1, "name": "Alice", "is_active": True})
    with pytest.raises(StrictDataclassFieldValidationError):
        validate_typed_dict(User, {"id": "1", "name": "Alice", "is_active": True})

def test_validate_typed_dict_non_total_supports_optional_fields():
    class User(TypedDict, total=False):
        id: int
        name: str

    validate_typed_dict(User, {"id": 1})
    validate_typed_dict(User, {"name": "Bob"})
    validate_typed_dict(User, {})

def test_setattr_after_init_triggers_validation():
    @dataclass
    @strict
    class TestDC:
        field: int

    obj = TestDC(field=1)
    with pytest.raises(StrictDataclassFieldValidationError):
        obj.field = "string"

def test_repr_includes_extra_fields_when_accept_kwargs_enabled():
    @dataclass(repr=True)
    @strict(accept_kwargs=True)
    class TestDC:
        id: int

    obj = TestDC(id=1, extra="value")
    assert "id=1" in repr(obj)
    assert "*extra='value'" in repr(obj)

import sys
import copy
from unittest import mock
import pytest
from . import capi as module_under_test

@pytest.fixture(autouse=True)
def reset_globals():
    original_outneeds = copy.deepcopy(module_under_test.outneeds)
    original_needs = copy.deepcopy(module_under_test.needs)
    original_cppmacros = copy.deepcopy(module_under_test.cppmacros)
    yield
    module_under_test.outneeds = original_outneeds
    module_under_test.needs = original_needs
    module_under_test.cppmacros = original_cppmacros

def test_errmess_writes_to_stderr_when_available():
    test_msg = "test error message\n"
    with mock.patch.object(sys.stderr, 'write') as mock_write:
        module_under_test.errmess(test_msg)
        mock_write.assert_called_once_with(test_msg)

def test_errmess_no_op_when_stderr_is_none():
    test_msg = "test error message\n"
    with mock.patch.object(sys, 'stderr', None):
        # Should not raise any error
        module_under_test.errmess(test_msg)

def test_append_needs_adds_single_known_need():
    assert 'math.h' not in module_under_test.outneeds['includes0']
    module_under_test.append_needs('math.h')
    assert 'math.h' in module_under_test.outneeds['includes0']

def test_append_needs_adds_list_of_needs():
    assert 'math.h' not in module_under_test.outneeds['includes0']
    assert 'string.h' not in module_under_test.outneeds['includes0']
    module_under_test.append_needs(['math.h', 'string.h'])
    assert 'math.h' in module_under_test.outneeds['includes0']
    assert 'string.h' in module_under_test.outneeds['includes0']

def test_append_needs_unknown_need_emits_error():
    unknown_need = "nonexistent_need"
    with mock.patch.object(sys.stderr, 'write') as mock_write:
        module_under_test.append_needs(unknown_need)
        mock_write.assert_called_once_with(f'append_needs: unknown need {repr(unknown_need)}\n')

def test_append_needs_handles_nested_dependencies():
    module_under_test.needs['test_need'] = ['math.h', 'string.h']
    module_under_test.cppmacros['test_need'] = '/* test_need */'
    module_under_test.append_needs('test_need')
    assert 'math.h' in module_under_test.outneeds['includes0']
    assert 'string.h' in module_under_test.outneeds['includes0']
    assert 'test_need' in module_under_test.outneeds['cppmacros']

def test_append_needs_flag_zero_returns_dependency_tree():
    module_under_test.needs['test_need'] = ['math.h']
    module_under_test.cppmacros['test_need'] = '/* test_need */'
    result = module_under_test.append_needs('test_need', flag=0)
    assert isinstance(result, dict)
    assert 'includes0' in result
    assert 'math.h' in result['includes0']
    assert 'cppmacros' in result
    assert 'test_need' in result['cppmacros']

def test_append_needs_ignores_duplicates():
    module_under_test.append_needs('math.h')
    initial_count = len(module_under_test.outneeds['includes0'])
    module_under_test.append_needs('math.h')
    assert len(module_under_test.outneeds['includes0']) == initial_count

def test_get_needs_returns_correctly_sorted_dependencies():
    module_under_test.needs['b'] = ['a']
    module_under_test.cppmacros['a'] = '/*a*/'
    module_under_test.cppmacros['b'] = '/*b*/'
    module_under_test.append_needs('b')
    res = module_under_test.get_needs()
    idx_a = res['cppmacros'].index('a')
    idx_b = res['cppmacros'].index('b')
    assert idx_a < idx_b

def test_get_needs_circular_dependency_emits_error():
    module_under_test.needs['a'] = ['b']
    module_under_test.needs['b'] = ['a']
    module_under_test.cppmacros['a'] = '/*a*/'
    module_under_test.cppmacros['b'] = '/*b*/'
    module_under_test.outneeds['cppmacros'] = ['a', 'b']
    with mock.patch.object(sys.stderr, 'write') as mock_write:
        res = module_under_test.get_needs()
        assert any('circular dependence' in call.args[0] for call in mock_write.call_args_list)
        assert 'a' in res['cppmacros']
        assert 'b' in res['cppmacros']

def test_get_needs_returns_default_for_empty_outneeds():
    module_under_test.outneeds['includes0'] = []
    res = module_under_test.get_needs()
    assert res['includes0'] == ['includes0']

def test_buildcfuncs_adds_cppmacros_for_c2capi_map():
    test_map = {'int': 'NPY_INT', 'float': 'NPY_FLOAT'}
    with mock.patch('module_under_test.capi_maps.c2capi_map', test_map):
        module_under_test.buildcfuncs()
        assert 'pyarr_from_p_int1' in module_under_test.cppmacros
        assert 'PyArray_SimpleNewFromData(0,NULL,NPY_INT,(char *)v)' in module_under_test.cppmacros['pyarr_from_p_int1']
        assert 'pyarr_from_p_float1' in module_under_test.cppmacros
        assert 'PyArray_SimpleNewFromData(0,NULL,NPY_FLOAT,(char *)v)' in module_under_test.cppmacros['pyarr_from_p_float1']

def test_buildcfuncs_adds_special_string_macro():
    with mock.patch('module_under_test.capi_maps.c2capi_map', {}):
        module_under_test.buildcfuncs()
        assert 'pyarr_from_p_string1' in module_under_test.cppmacros
        assert 'PyArray_New(&PyArray_Type, 1, dims, NPY_STRING' in module_under_test.cppmacros['pyarr_from_p_string1']

# Test helper function has_body_headers
def test_has_body_headers():
    # Normal case with content-length
    req1 = Mock(headers=[(b'Content-Length', b'100')])
    assert has_body_headers(req1) is True
    # Normal case with transfer-encoding
    req2 = Mock(headers=[(b'Transfer-Encoding', b'chunked')])
    assert has_body_headers(req2) is True
    # Boundary case no body headers
    req3 = Mock(headers=[(b'Host', b'example.com'), (b'User-Agent', b'test')])
    assert has_body_headers(req3) is False
    # Case-insensitive check
    req4 = Mock(headers=[(b'CONTENT-length', b'50')])
    assert has_body_headers(req4) is True

# Test connection initialization
def test_connection_init(mock_origin, mock_stream):
    conn = AsyncHTTP2Connection(origin=mock_origin, stream=mock_stream, keepalive_expiry=30.0)
    assert conn._origin == mock_origin
    assert conn._state == HTTPConnectionState.IDLE
    assert conn._sent_connection_init is False
    assert conn._request_count == 0
    assert conn._used_all_stream_ids is False
    assert conn._connection_error is False

# Test can_handle_request
def test_can_handle_request(h2_connection, mock_origin):
    # Same origin
    assert h2_connection.can_handle_request(mock_origin) is True
    # Different scheme
    wrong_scheme = Origin(scheme=b'http', host=b'example.com', port=443)
    assert h2_connection.can_handle_request(wrong_scheme) is False
    # Different host
    wrong_host = Origin(scheme=b'https', host=b'example.org', port=443)
    assert h2_connection.can_handle_request(wrong_host) is False
    # Different port
    wrong_port = Origin(scheme=b'https', host=b'example.com', port=8443)
    assert h2_connection.can_handle_request(wrong_port) is False

# Test state check methods
def test_state_checks(h2_connection):
    assert h2_connection.is_idle() is True
    assert h2_connection.is_available() is True
    assert h2_connection.is_closed() is False
    assert h2_connection.has_expired() is False
    # Test expired state
    h2_connection._expire_at = time.monotonic() - 10
    assert h2_connection.has_expired() is True
    # Test closed state
    asyncio.run(h2_connection.aclose())
    assert h2_connection.is_closed() is True
    assert h2_connection.is_available() is False

# Test normal GET request flow
@pytest.mark.asyncio
async def test_handle_get_request(h2_connection, test_request, mock_stream):
    # Mock events
    response_event = h2.events.ResponseReceived()
    response_event.stream_id = 1
    response_event.headers = [(b':status', b'200'), (b'content-type', b'text/plain')]
    data_event = h2.events.DataReceived()
    data_event.stream_id = 1
    data_event.data = b'hello world'
    data_event.flow_controlled_length = 11
    end_event = h2.events.StreamEnded()
    end_event.stream_id = 1
    h2_connection._h2_state.receive_data.return_value = [response_event, data_event, end_event]
    mock_stream.read.side_effect = [b'', b'']

    response = await h2_connection.handle_async_request(test_request)
    assert response.status == 200
    assert dict(response.headers) == {b'content-type': b'text/plain'}
    assert response.extensions[b'http_version'] == b'HTTP/2'
    # Read response body
    body = b''
    async for chunk in response.content:
        body += chunk
    assert body == b'hello world'
    # Verify write calls
    assert mock_stream.write.called
    assert h2_connection._h2_state.send_headers.called
    assert h2_connection._request_count == 1

# Test remote settings change
@pytest.mark.asyncio
async def test_remote_settings_change(h2_connection):
    h2_connection._sent_connection_init = True
    h2_connection._max_streams = 100
    h2_connection._max_streams_semaphore = AsyncSemaphore(100)
    # Simulate max concurrent streams reduce
    settings_event = h2.events.RemoteSettingsChanged()
    settings_event.changed_settings = {
        h2.settings.SettingCodes.MAX_CONCURRENT_STREAMS: Mock(new_value=50)
    }
    await h2_connection._receive_remote_settings_change(settings_event)
    assert h2_connection._max_streams == 50
    # Simulate max concurrent streams increase
    settings_event.changed_settings = {
        h2.settings.SettingCodes.MAX_CONCURRENT_STREAMS: Mock(new_value=75)
    }
    await h2_connection._receive_remote_settings_change(settings_event)
    assert h2_connection._max_streams == 75

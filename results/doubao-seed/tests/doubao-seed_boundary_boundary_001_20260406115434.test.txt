import pytest
from pygments.token import *
from pygments.testutil import assertTokens
from pygments.lexers.python import (
    PythonLexer, Python2Lexer, PythonConsoleLexer,
    PythonTracebackLexer, Python2TracebackLexer,
    CythonLexer, DgLexer, NumPyLexer
)


class TestPythonLexer:
    def test_basic_syntax(self):
        code = "def hello_world(name: str) -> str:\n    return f'Hello {name}!'"
        assertTokens(PythonLexer(), code, [
            (Keyword, 'def'),
            (Whitespace, ' '),
            (Name.Function, 'hello_world'),
            (Punctuation, '('),
            (Name, 'name'),
            (Punctuation, ':'),
            (Whitespace, ' '),
            (Name.Builtin, 'str'),
            (Punctuation, ')'),
            (Whitespace, ' '),
            (Operator, '->'),
            (Whitespace, ' '),
            (Name.Builtin, 'str'),
            (Punctuation, ':'),
            (Whitespace, '\n    '),
            (Keyword, 'return'),
            (Whitespace, ' '),
            (String.Affix, 'f'),
            (String.Single, "'"),
            (String.Single, 'Hello '),
            (String.Interpol, '{name}'),
            (String.Single, '!'),
            (String.Single, "'"),
        ])

    def test_docstring_detection(self):
        code = '"""Module level docstring"""\nclass Test:\n    \'\'\'Class docstring\'\'\'\n    pass'
        assertTokens(PythonLexer(), code, [
            (String.Affix, ''),
            (String.Doc, '"""Module level docstring"""'),
            (Whitespace, '\n'),
            (Keyword, 'class'),
            (Whitespace, ' '),
            (Name.Class, 'Test'),
            (Punctuation, ':'),
            (Whitespace, '\n    '),
            (String.Affix, ''),
            (String.Doc, "'''Class docstring'''"),
            (Whitespace, '\n    '),
            (Keyword, 'pass'),
        ])

    def test_number_tokenization(self):
        code = '123 0x1A 0b101 0o755 1.23e4 3.14j'
        assertTokens(PythonLexer(), code, [
            (Number.Integer, '123'),
            (Whitespace, ' '),
            (Number.Hex, '0x1A'),
            (Whitespace, ' '),
            (Number.Bin, '0b101'),
            (Whitespace, ' '),
            (Number.Oct, '0o755'),
            (Whitespace, ' '),
            (Number.Float, '1.23e4'),
            (Whitespace, ' '),
            (Number.Float, '3.14j'),
        ])

    def test_soft_keywords(self):
        code = 'match value:\n    case 1 | 2:\n        match = 3'
        assertTokens(PythonLexer(), code, [
            (Keyword, 'match'),
            (Whitespace, ' '),
            (Name, 'value'),
            (Punctuation, ':'),
            (Whitespace, '\n    '),
            (Keyword, 'case'),
            (Whitespace, ' '),
            (Number.Integer, '1'),
            (Whitespace, ' '),
            (Operator, '|'),
            (Whitespace, ' '),
            (Number.Integer, '2'),
            (Punctuation, ':'),
            (Whitespace, '\n        '),
            (Name, 'match'),
            (Whitespace, ' '),
            (Operator, '='),
            (Whitespace, ' '),
            (Number.Integer, '3'),
        ])

    def test_analyse_text(self):
        assert PythonLexer.analyse_text('#!/usr/bin/env python3') > 0
        assert PythonLexer.analyse_text('import sys\nprint("test")') > 0
        assert PythonLexer.analyse_text('<?php echo "not python"') == 0


class TestPython2Lexer:
    def test_python2_specific_features(self):
        code = 'print "Hello World"\nx = 123L'
        assertTokens(Python2Lexer(), code, [
            (Keyword, 'print'),
            (Whitespace, ' '),
            (String.Affix, ''),
            (String.Double, '"Hello World"'),
            (Whitespace, '\n'),
            (Name, 'x'),
            (Whitespace, ' '),
            (Operator, '='),
            (Whitespace, ' '),
            (Number.Integer.Long, '123L'),
        ])

    def test_analyse_text(self):
        assert Python2Lexer.analyse_text('#!/usr/bin/env python2.7') > 0
        assert Python2Lexer.analyse_text('#!/usr/bin/env python3') == 0


class TestPythonConsoleLexer:
    def test_basic_pycon_session(self):
        session = '>>> def add(a, b):\n...     return a + b\n... \n>>> add(2, 3)\n5'
        lexer = PythonConsoleLexer(python3=True)
        assert next(lexer.get_tokens(session)) == (Generic.Prompt, '>>> ')


class TestPythonTracebackLexer:
    def test_traceback_parsing(self):
        tb = '''Traceback (most recent call last):
  File "test.py", line 3, in <module>
    raise ValueError("test error")
ValueError: test error
'''
        assertTokens(PythonTracebackLexer(), tb, [
            (Generic.Traceback, 'Traceback (most recent call last):\n'),
            (Text, '  File '),
            (Name.Builtin, '"test.py"'),
            (Text, ', line '),
            (Number, '3'),
            (Text, ', in '),
            (Name, '<module>'),
            (Whitespace, '\n'),
            (Whitespace, '    '),
            (Keyword, 'raise'),
            (Whitespace, ' '),
            (Name.Exception, 'ValueError'),
            (Punctuation, '('),
            (String.Double, '"test error"'),
            (Punctuation, ')'),
            (Whitespace, '\n'),
            (Generic.Error, 'ValueError'),
            (Text, ': '),
            (Name, 'test error'),
            (Whitespace, '\n'),
        ])

class TestNumPyLexer:
    def test_numpy_keyword_detection(self):
        code = 'import numpy as np\narr = np.array([1,2,3])'
        assertTokens(NumPyLexer(), code, [
            (Keyword.Namespace, 'import'),
            (Whitespace, ' '),
            (Name.Namespace, 'numpy'),
            (Whitespace, ' '),
            (Keyword, 'as'),
            (Whitespace, ' '),
            (Name.Namespace, 'np'),
            (Whitespace, '\n'),
            (Name, 'arr'),
            (Whitespace, ' '),
            (Operator, '='),
            (Whitespace, ' '),
            (Name, 'np'),
            (Operator, '.'),
            (Keyword.Pseudo, 'array'),
            (Punctuation, '('),
            (Punctuation, '['),
            (Number.Integer, '1'),
            (Punctuation, ','),
            (Number.Integer, '2'),
            (Punctuation, ','),
            (Number.Integer, '3'),
            (Punctuation, ']'),
            (Punctuation, ')'),
        ])

    def test_analyse_text(self):
        assert NumPyLexer.analyse_text('import numpy as np\nx = np.array([1,2,3])') > 0
        assert NumPyLexer.analyse_text('import sys\nprint("no numpy")') == 0


class TestEdgeCases:
    def test_empty_input(self):
        assert list(PythonLexer().get_tokens('')) == []

    def test_all_whitespace(self):
        assertTokens(PythonLexer(), '   \n\t\n  ', [
            (Whitespace, '   '),
            (Whitespace, '\n'),
            (Whitespace, '\t'),
            (Whitespace, '\n'),
            (Whitespace, '  '),
        ])

    def test_raw_byte_strings(self):
        code = r"rb'hello\x00world' u'unicode string'"
        assertTokens(PythonLexer(), code, [
            (String.Affix, 'rb'),
            (String.Single, "'"),
            (String.Escape, r'\x00'),
            (String.Single, "hello\\x00world'"),
            (Whitespace, ' '),
            (String.Affix, 'u'),
            (String.Single, "'unicode string'"),
        ])

class TestHasBodyHeaders(unittest.TestCase):
    def test_with_content_length(self):
        request = Mock(headers=[(b'Content-Length', b'100')])
        self.assertTrue(has_body_headers(request))
    
    def test_with_transfer_encoding(self):
        request = Mock(headers=[(b'Transfer-Encoding', b'chunked')])
        self.assertTrue(has_body_headers(request))
    
    def test_without_body_headers(self):
        request = Mock(headers=[(b'Host', b'example.com'), (b'Accept', b'*/*')])
        self.assertFalse(has_body_headers(request))
    
    def test_mixed_case_header_names(self):
        request = Mock(headers=[(b'CONTENT-length', b'200')])
        self.assertTrue(has_body_headers(request))

class TestHTTP2Connection(unittest.TestCase):
    def setUp(self):
        self.origin = Origin(scheme=b'https', host=b'example.com', port=443)
        self.mock_stream = Mock(spec=NetworkStream)
        self.keepalive_expiry = 30.0
        with patch('httpcore.http2.h2.connection.H2Connection') as mock_h2_conn:
            self.mock_h2_state = Mock()
            self.mock_h2_state.get_next_available_stream_id.return_value = 1
            self.mock_h2_state.local_settings = Mock(max_concurrent_streams=100)
            self.mock_h2_state.max_outbound_frame_size = 16384
            self.mock_h2_state.local_flow_control_window.return_value = 65535
            self.mock_h2_state.data_to_send.return_value = b''
            self.mock_h2_state.state_machine.state = Mock()
            mock_h2_conn.return_value = self.mock_h2_state
            self.conn = HTTP2Connection(
                origin=self.origin,
                stream=self.mock_stream,
                keepalive_expiry=self.keepalive_expiry
            )

    def test_initialization(self):
        self.assertEqual(self.conn._origin, self.origin)
        self.assertEqual(self.conn._network_stream, self.mock_stream)
        self.assertEqual(self.conn._keepalive_expiry, 30.0)
        self.assertEqual(self.conn._state, HTTPConnectionState.IDLE)
        self.assertEqual(self.conn._request_count, 0)
        self.assertFalse(self.conn._sent_connection_init)

    def test_can_handle_request(self):
        matching_origin = Origin(b'https', b'example.com', 443)
        non_matching_origin = Origin(b'https', b'other.com', 443)
        self.assertTrue(self.conn.can_handle_request(matching_origin))
        self.assertFalse(self.conn.can_handle_request(non_matching_origin))

    def test_handle_request_wrong_origin_raises(self):
        wrong_origin = Origin(b'https', b'wrong.com', 443)
        mock_url = Mock(origin=wrong_origin)
        request = Mock(url=mock_url)
        with self.assertRaises(RuntimeError):
            self.conn.handle_request(request)

    def test_handle_request_closed_connection_raises(self):
        self.conn._state = HTTPConnectionState.CLOSED
        mock_url = Mock(origin=self.origin)
        request = Mock(url=mock_url)
        with self.assertRaises(ConnectionNotAvailable):
            self.conn.handle_request(request)

    def test_handle_request_no_available_stream_ids_raises(self):
        self.conn._state = HTTPConnectionState.ACTIVE
        self.conn._sent_connection_init = True
        self.conn._max_streams_semaphore = Mock(spec=Semaphore)
        self.mock_h2_state.get_next_available_stream_id.side_effect = h2.exceptions.NoAvailableStreamIDError()
        mock_url = Mock(origin=self.origin)
        request = Mock(url=mock_url, headers=[(b'Host', b'example.com')], extensions={})
        initial_request_count = self.conn._request_count
        with self.assertRaises(ConnectionNotAvailable):
            self.conn.handle_request(request)
        self.assertEqual(self.conn._request_count, initial_request_count)
        self.assertTrue(self.conn._used_all_stream_ids)

    def test_handle_request_normal_path(self):
        self.conn._state = HTTPConnectionState.IDLE
        mock_url = Mock(origin=self.origin, scheme=b'https', target=b'/test')
        request = Request(
            method=b'GET',
            url=mock_url,
            headers=[(b'Host', b'example.com'), (b'Accept', b'*/*')],
            stream=iter([]),
            extensions={}
        )
        mock_response_event = Mock(spec=h2.events.ResponseReceived)
        mock_response_event.headers = [(b':status', b'200'), (b'Content-Type', b'text/plain')]
        self.conn._receive_response = Mock(return_value=(200, [(b'Content-Type', b'text/plain')]))
        self.conn._send_connection_init = Mock()
        self.conn._send_request_headers = Mock()
        self.conn._send_request_body = Mock()

        response = self.conn.handle_request(request)
        self.assertEqual(response.status, 200)
        self.assertEqual(response.headers, [(b'Content-Type', b'text/plain')])
        self.assertEqual(response.extensions['http_version'], b'HTTP/2')
        self.assertEqual(self.conn._state, HTTPConnectionState.ACTIVE)
        self.assertEqual(self.conn._request_count, 1)
        self.assertTrue(self.conn._sent_connection_init)

    def test_receive_remote_settings_increase(self):
        self.conn._max_streams = 1
        self.conn._max_streams_semaphore = Mock(spec=Semaphore)
        mock_event = Mock(spec=h2.events.RemoteSettingsChanged)
        mock_setting = Mock(new_value=50)
        mock_event.changed_settings = {h2.settings.SettingCodes.MAX_CONCURRENT_STREAMS: mock_setting}
        self.conn._receive_remote_settings_change(mock_event)
        self.assertEqual(self.conn._max_streams, 50)
        self.assertEqual(self.conn._max_streams_semaphore.release.call_count, 49)

    def test_receive_remote_settings_decrease(self):
        self.conn._max_streams = 100
        self.conn._max_streams_semaphore = Mock(spec=Semaphore)
        mock_event = Mock(spec=h2.events.RemoteSettingsChanged)
        mock_setting = Mock(new_value=20)
        mock_event.changed_settings = {h2.settings.SettingCodes.MAX_CONCURRENT_STREAMS: mock_setting}
        self.conn._receive_remote_settings_change(mock_event)
        self.assertEqual(self.conn._max_streams, 20)
        self.assertEqual(self.conn._max_streams_semaphore.acquire.call_count, 80)

    def test_is_available(self):
        self.assertTrue(self.conn.is_available())
        self.conn._state = HTTPConnectionState.CLOSED
        self.assertFalse(self.conn.is_available())
        self.conn._state = HTTPConnectionState.ACTIVE
        self.conn._connection_error = True
        self.assertFalse(self.conn.is_available())
        self.conn._connection_error = False
        self.conn._used_all_stream_ids = True
        self.assertFalse(self.conn.is_available())

    @patch('time.monotonic')
    def test_has_expired(self, mock_monotonic):
        mock_monotonic.return_value = 1000.0
        self.conn._expire_at = 999.0
        self.assertTrue(self.conn.has_expired())
        self.conn._expire_at = 1001.0
        self.assertFalse(self.conn.has_expired())
        self.conn._expire_at = None
        self.assertFalse(self.conn.has_expired())

    def test_state_checks(self):
        self.conn._state = HTTPConnectionState.IDLE
        self.assertTrue(self.conn.is_idle())
        self.assertFalse(self.conn.is_closed())
        self.conn._state = HTTPConnectionState.CLOSED
        self.assertFalse(self.conn.is_idle())
        self.assertTrue(self.conn.is_closed())

    def test_close(self):
        self.conn.close()
        self.mock_h2_state.close_connection.assert_called_once()
        self.mock_stream.close.assert_called_once()
        self.assertEqual(self.conn._state, HTTPConnectionState.CLOSED)

    def test_info_repr(self):
        info_str = self.conn.info()
        self.assertIn('https://example.com:443', info_str)
        self.assertIn('IDLE', info_str)
        self.assertIn('Request Count: 0', info_str)
        repr_str = repr(self.conn)
        self.assertIn('HTTP2Connection', repr_str)
        self.assertIn('https://example.com:443', repr_str)
        self.assertIn('IDLE', repr_str)

    def test_context_manager(self):
        with patch.object(self.conn, 'close') as mock_close:
            with self.conn as conn:
                self.assertEqual(conn, self.conn)
            mock_close.assert_called_once()

class TestHTTP2ConnectionByteStream(unittest.TestCase):
    def setUp(self):
        self.mock_conn = Mock(spec=HTTP2Connection)
        self.mock_request = Mock()
        self.stream_id = 1
        self.stream = HTTP2ConnectionByteStream(
            connection=self.mock_conn,
            request=self.mock_request,
            stream_id=self.stream_id
        )

    def test_iter_yields_chunks(self):
        test_chunks = [b'chunk1', b'chunk2']
        self.mock_conn._receive_response_body.return_value = iter(test_chunks)
        chunks = list(iter(self.stream))
        self.assertEqual(chunks, test_chunks)
        self.mock_conn._receive_response_body.assert_called_once_with(request=self.mock_request, stream_id=1)

    def test_close(self):
        self.stream.close()
        self.assertTrue(self.stream._closed)
        self.mock_conn._response_closed.assert_called_once_with(stream_id=1)

    def test_iter_error_closes_stream(self):
        self.mock_conn._receive_response_body.side_effect = RemoteProtocolError('test error')
        with patch.object(self.stream, 'close') as mock_close:
            with self.assertRaises(RemoteProtocolError):
                list(iter(self.stream))
            mock_close.assert_called_once()

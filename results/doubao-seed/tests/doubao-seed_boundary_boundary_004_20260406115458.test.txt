import pytest
from datetime import datetime, timedelta, date, time
from dateutil import tz
from boundary import isoparse, isoparser


def test_valid_common_date_parsing():
    assert isoparse('2023') == datetime(2023, 1, 1)
    assert isoparse('2023-10') == datetime(2023, 10, 1)
    assert isoparse('2023-10-05') == datetime(2023, 10, 5)
    assert isoparse('20231005') == datetime(2023, 10, 5)
    assert isoparse('2020-02-29') == datetime(2020, 2, 29)
    assert isoparse('1900-02-28') == datetime(1900, 2, 28)
    assert isoparse('9999-12-31') == datetime(9999, 12, 31)


def test_valid_uncommon_date_parsing():
    assert isoparse('2023-001') == datetime(2023, 1, 1)
    assert isoparse('2023365') == datetime(2023, 12, 31)
    assert isoparse('2020-366') == datetime(2020, 12, 31)
    assert isoparse('2023-W01') == datetime(2023, 1, 2)
    assert isoparse('2023W011') == datetime(2023, 1, 2)
    assert isoparse('2023-W01-7') == datetime(2023, 1, 8)
    assert isoparse('2023W017') == datetime(2023, 1, 8)


def test_valid_time_parsing():
    assert isoparse('2023-10-05T00:00:00') == datetime(2023, 10, 5, 0, 0, 0)
    assert isoparse('2023-10-05T12:34:56') == datetime(2023, 10, 5, 12, 34, 56)
    assert isoparse('20231005T123456') == datetime(2023, 10, 5, 12, 34, 56)
    assert isoparse('2023-10-05T12:34:56.123') == datetime(2023, 10, 5, 12, 34, 56, 123000)
    assert isoparse('2023-10-05T12:34:56,123456') == datetime(2023, 10, 5, 12, 34, 56, 123456)
    assert isoparse('2023-10-05T24:00:00') == datetime(2023, 10, 6, 0, 0, 0)


def test_valid_timezone_parsing():
    assert isoparse('2023-10-05T12:34:56Z') == datetime(2023, 10, 5, 12, 34, 56, tzinfo=tz.UTC)
    assert isoparse('2023-10-05T12:34:56z') == datetime(2023, 10, 5, 12, 34, 56, tzinfo=tz.UTC)
    assert isoparse('2023-10-05T12:34+01') == datetime(2023, 10, 5, 12, 34, tzinfo=tz.tzoffset(None, 3600))
    assert isoparse('2023-10-05T12:34+01:00') == datetime(2023, 10, 5, 12, 34, tzinfo=tz.tzoffset(None, 3600))
    assert isoparse('2023-10-05T12:34-0530') == datetime(2023, 10, 5, 12, 34, tzinfo=tz.tzoffset(None, -19800))
    assert isoparse('2023-10-05T12:34+00:00') == datetime(2023, 10, 5, 12, 34, tzinfo=tz.UTC)


def test_custom_separator():
    space_parser = isoparser(sep=' ')
    assert space_parser.isoparse('2023-10-05 12:34:56') == datetime(2023, 10, 5, 12, 34, 56)
    underscore_parser = isoparser(sep='_')
    assert underscore_parser.isoparse('20231005_123456') == datetime(2023, 10, 5, 12, 34, 56)


def test_standalone_parse_methods():
    parser = isoparser()
    assert parser.parse_isodate('2023-10-05') == date(2023, 10, 5)
    assert parser.parse_isotime('12:34:56') == time(12, 34, 56)
    assert parser.parse_isotime('24:00:00') == time(0, 0, 0)
    assert parser.parse_tzstr(b'+05:30').utcoffset(None) == timedelta(hours=5, minutes=30)


def test_invalid_separator_initialization():
    with pytest.raises(ValueError):
        isoparser(sep='12')
    with pytest.raises(ValueError):
        isoparser(sep='5')
    with pytest.raises(ValueError):
        isoparser(sep='ñ')
    with pytest.raises(ValueError):
        isoparser(sep='')


def test_invalid_date_inputs():
    with pytest.raises(ValueError):
        isoparse('202')
    with pytest.raises(ValueError):
        isoparse('2023-13')
    with pytest.raises(ValueError):
        isoparse('2023-02-30')
    with pytest.raises(ValueError):
        isoparse('2023-W54')
    with pytest.raises(ValueError):
        isoparse('2023-W01-8')
    with pytest.raises(ValueError):
        isoparse('2023-367')
    with pytest.raises(ValueError):
        isoparse('2023-1005')
    with pytest.raises(ValueError):
        isoparse('202310-05')
    with pytest.raises(ValueError):
        isoparse('2023-10-05ä')


def test_invalid_time_inputs():
    with pytest.raises(ValueError):
        isoparse('2023-10-05T1')
    with pytest.raises(ValueError):
        isoparse('2023-10-05T25:00:00')
    with pytest.raises(ValueError):
        isoparse('2023-10-05T12:60:00')
    with pytest.raises(ValueError):
        isoparse('2023-10-05T12:34:60')
    with pytest.raises(ValueError):
        isoparse('2023-10-05T24:00:01')
    with pytest.raises(ValueError):
        isoparse('2023-10-05T1234:56')
    with pytest.raises(ValueError):
        isoparse('2023-10-05T12:3456')


def test_invalid_timezone_inputs():
    with pytest.raises(ValueError):
        isoparse('2023-10-05T12:34+25:00')
    with pytest.raises(ValueError):
        isoparse('2023-10-05T12:34-00:61')
    with pytest.raises(ValueError):
        isoparse('2023-10-05T12:340000')
    with pytest.raises(ValueError):
        isoparse('2023-10-05T12:34A0100')


def test_standalone_method_exceptions():
    parser = isoparser()
    with pytest.raises(ValueError):
        parser.parse_isodate('2023-10-05T12:00')
    with pytest.raises(ValueError):
        parser.parse_isotime('24:00:01')
    with pytest.raises(ValueError):
        parser.parse_tzstr(b'0000')
    with pytest.raises(ValueError):
        parser.parse_tzstr(b'+2500')

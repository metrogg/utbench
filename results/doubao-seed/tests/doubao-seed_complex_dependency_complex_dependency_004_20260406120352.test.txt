class TestExtractManager(unittest.TestCase):
    def setUp(self):
        self.test_cache_dir = tempfile.gettempdir()
        self.test_input_path = "/test/path/archive.tar.gz"
        self.test_output_path = "/test/cache/extracted/hash123"
        with patch("your_module.config.EXTRACTED_DATASETS_DIR", "extracted"), \
             patch("your_module.config.EXTRACTED_DATASETS_PATH", "/default/extracted"):
            self.manager_with_cache = ExtractManager(cache_dir=self.test_cache_dir)
            self.manager_no_cache = ExtractManager()

    def test_init(self):
        self.assertEqual(self.manager_with_cache.extract_dir, os.path.join(self.test_cache_dir, "extracted"))
        self.assertEqual(self.manager_no_cache.extract_dir, "/default/extracted")
        self.assertEqual(self.manager_with_cache.extractor, Extractor)

    @patch("your_module.extractors.hash_url_to_filename")
    def test_get_output_path(self, mock_hash):
        mock_hash.return_value = "hash123"
        expected = os.path.join(self.manager_with_cache.extract_dir, "hash123")
        result = self.manager_with_cache._get_output_path(self.test_input_path)
        mock_hash.assert_called_once_with(os.path.abspath(self.test_input_path))
        self.assertEqual(result, expected)

    @patch("os.path.isfile")
    @patch("os.path.isdir")
    @patch("os.listdir")
    def test_do_extract(self, mock_listdir, mock_isdir, mock_isfile):
        self.assertTrue(self.manager_with_cache._do_extract(self.test_output_path, force_extract=True))
        mock_isfile.return_value = False
        mock_isdir.return_value = False
        self.assertTrue(self.manager_with_cache._do_extract(self.test_output_path, force_extract=False))
        mock_isdir.return_value = True
        mock_listdir.return_value = []
        self.assertTrue(self.manager_with_cache._do_extract(self.test_output_path, force_extract=False))
        mock_listdir.return_value = ["file1.txt"]
        self.assertFalse(self.manager_with_cache._do_extract(self.test_output_path, force_extract=False))
        mock_isfile.return_value = True
        self.assertFalse(self.manager_with_cache._do_extract(self.test_output_path, force_extract=False))

    @patch.object(ExtractManager, "_do_extract")
    @patch.object(ExtractManager, "_get_output_path")
    @patch.object(Extractor, "infer_extractor_format")
    @patch.object(Extractor, "extract")
    def test_extract(self, mock_extract, mock_infer, mock_get_output, mock_do_extract):
        mock_infer.return_value = None
        result = self.manager_with_cache.extract(self.test_input_path)
        self.assertEqual(result, self.test_input_path)
        mock_extract.assert_not_called()

        mock_infer.return_value = "gzip"
        mock_get_output.return_value = self.test_output_path
        mock_do_extract.return_value = True
        result = self.manager_with_cache.extract(self.test_input_path, force_extract=True)
        self.assertEqual(result, self.test_output_path)
        mock_extract.assert_called_once_with(self.test_input_path, self.test_output_path, "gzip")

        mock_extract.reset_mock()
        mock_do_extract.return_value = False
        result = self.manager_with_cache.extract(self.test_input_path)
        self.assertEqual(result, self.test_output_path)
        mock_extract.assert_not_called()

class TestMagicNumberBaseExtractor(unittest.TestCase):
    class TestExtractor(MagicNumberBaseExtractor):
        magic_numbers = [b"\x00\x01", b"\x02\x03\x04"]
        def extract(*args, **kwargs): pass
        def is_extractable(*args, **kwargs): return super().is_extractable(*args, **kwargs)

    def test_read_magic_number(self):
        test_content = b"\x00\x01\x02\x03\x04\x05"
        with patch("builtins.open", mock_open(read_data=test_content)):
            result = self.TestExtractor.read_magic_number("test/path", 3)
            self.assertEqual(result, b"\x00\x01\x02")

    def test_is_extractable(self):
        test_content = b"\x00\x01\xFF\xFF"
        with patch("builtins.open", mock_open(read_data=test_content)):
            self.assertTrue(self.TestExtractor.is_extractable("test/path"))
        test_content = b"\x02\x03\x04\xFF"
        with patch("builtins.open", mock_open(read_data=test_content)):
            self.assertTrue(self.TestExtractor.is_extractable("test/path"))
        test_content = b"\xFF\xFF\xFF"
        with patch("builtins.open", mock_open(read_data=test_content)):
            self.assertFalse(self.TestExtractor.is_extractable("test/path"))
        with patch("builtins.open", side_effect=OSError("File not found")):
            self.assertFalse(self.TestExtractor.is_extractable("test/path"))
        self.assertTrue(self.TestExtractor.is_extractable("test/path", magic_number=b"\x00\x01\xFF"))
        self.assertFalse(self.TestExtractor.is_extractable("test/path", magic_number=b"\xFF\x00\x01"))

class TestTarExtractor(unittest.TestCase):
    @patch("tarfile.is_tarfile")
    def test_is_extractable(self, mock_is_tar):
        mock_is_tar.return_value = True
        self.assertTrue(TarExtractor.is_extractable("test.tar"))
        mock_is_tar.return_value = False
        self.assertFalse(TarExtractor.is_extractable("test.txt"))

    def test_safemembers(self):
        output_path = "/test/output"
        good_member = MagicMock(name="good.txt")
        good_member.name = "good.txt"
        good_member.issym.return_value = False
        good_member.islnk.return_value = False
        bad_path_member = MagicMock(name="../bad.txt")
        bad_path_member.name = "../bad.txt"
        bad_path_member.issym.return_value = False
        bad_path_member.islnk.return_value = False
        bad_sym_member = MagicMock(name="sym.txt")
        bad_sym_member.name = "sym.txt"
        bad_sym_member.issym.return_value = True
        bad_sym_member.linkname = "../../etc/passwd"
        members = [good_member, bad_path_member, bad_sym_member]
        result = list(TarExtractor.safemembers(members, output_path))
        self.assertEqual(len(result), 1)
        self.assertEqual(result[0].name, "good.txt")

    @patch("tarfile.open")
    @patch("os.makedirs")
    def test_extract(self, mock_makedirs, mock_tar_open):
        mock_tar = MagicMock()
        mock_tar_open.return_value = mock_tar
        test_input = "test.tar"
        test_output = "/test/output"
        TarExtractor.extract(test_input, test_output)
        mock_makedirs.assert_called_once_with(test_output, exist_ok=True)
        mock_tar_open.assert_called_once_with(test_input)
        mock_tar.extractall.assert_called_once()
        mock_tar.close.assert_called_once()

class TestGzipExtractor(unittest.TestCase):

        mock_file_open.assert_called_once_with(test_output, "wb")
        mock_copy.assert_called_once_with(mock_gzip, mock_file_open.return_value.__enter__.return_value)

class TestExtractor(unittest.TestCase):
    def test_get_magic_number_max_length(self):
        max_len = Extractor._get_magic_number_max_length()
        self.assertEqual(max_len, 6)

    @patch.object(MagicNumberBaseExtractor, "read_magic_number")
    def test_read_magic_number(self, mock_read):
        mock_read.return_value = b"\x1f\x8b"
        result = Extractor._read_magic_number("test.gz", 2)
        self.assertEqual(result, b"\x1f\x8b")
        mock_read.side_effect = OSError()
        result = Extractor._read_magic_number("test.gz", 2)
        self.assertEqual(result, b"")

    def test_infer_extractor_format(self):
        with patch.object(Extractor, "_read_magic_number", return_value=b"\x1f\x8b\x08"):
            self.assertEqual(Extractor.infer_extractor_format("test.gz"), "gzip")
        with patch.object(Extractor, "_read_magic_number", return_value=b"\x00\x00\x00"):
            self.assertIsNone(Extractor.infer_extractor_format("test.unknown"))

    @patch.object(Extractor, "infer_extractor_format")
    def test_is_extractable_deprecated(self, mock_infer):
        mock_infer.return_value = "gzip"
        with self.assertWarns(FutureWarning):
            result = Extractor.is_extractable("test.gz")
            self.assertTrue(result)
        with self.assertWarns(FutureWarning):
            result, extractor = Extractor.is_extractable("test.gz", return_extractor=True)
            self.assertTrue(result)
            self.assertEqual(extractor, GzipExtractor)
        mock_infer.return_value = None
        with self.assertWarns(FutureWarning):
            result = Extractor.is_extractable("test.txt")
            self.assertFalse(result)

    @patch("shutil.rmtree")
    @patch.object(TarExtractor, "extract")
    @patch("your_module.extractors.FileLock")
    @patch("os.makedirs")
    def test_extract(self, mock_makedirs, mock_lock, mock_tar_extract, mock_rmtree):
        mock_lock_instance = MagicMock()
        mock_lock.return_value.__enter__.return_value = mock_lock_instance
        test_input = "test.tar"
        test_output = "/test/output/test"
        Extractor.extract(test_input, test_output, "tar")
        mock_makedirs.assert_called_once_with(os.path.dirname(test_output), exist_ok=True)
        mock_lock.assert_called_once_with(str(Path(test_output).with_suffix(".lock")))
        mock_rmtree.assert_called_once_with(test_output, ignore_errors=True)
        mock_tar_extract.assert_called_once_with(test_input, test_output)

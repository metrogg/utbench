# Test 3: Class validators (methods starting with validate_)
def test_class_validators():
    @strict
    @dataclass
    class Account:
        balance: float
        min_balance: float = 0.0
        
        def validate_balance_positive(self):
            if self.balance < self.min_balance:
                raise ValueError(f"Balance {self.balance} cannot be below minimum {self.min_balance}")
        
        def validate_reasonable_amount(self):
            if self.balance > 1000000:
                raise ValueError("Balance too high")
    
    # Valid account
    acc = Account(balance=1000.0, min_balance=-100.0)
    assert acc.balance == 1000.0
    
    # Should fail class validation
    try:
        Account(balance=-50.0, min_balance=0.0)
        assert False, "Should have raised StrictDataclassClassValidationError"
    except StrictDataclassClassValidationError as e:
        assert "validate_balance_positive" in str(e.validator)
        assert isinstance(e.cause, ValueError)
    
    # Should fail second validator
    try:
        Account(balance=2000000.0)
        assert False, "Should have raised StrictDataclassClassValidationError"
    except StrictDataclassClassValidationError as e:
        assert "validate_reasonable_amount" in str(e.validator)
    
    # Invalid validator signature
    try:
        @strict
        @dataclass
        class InvalidValidator:
            x: int
            
            def validate_invalid(self, extra_arg):
                pass
        
        InvalidValidator(x=1)
        assert False, "Should have raised StrictDataclassDefinitionError"
    except StrictDataclassDefinitionError as e:
        assert "takes more than one argument" in str(e)

# Test 4: validated_field and custom validators
def test_validated_field():
    def is_positive(value):
        if value <= 0:
            raise ValueError("Value must be positive")
    
    def is_even(value):
        if value % 2 != 0:
            raise ValueError("Value must be even")
    
    @strict
    @dataclass
    class Numbers:
        positive_int: int = validated_field(is_positive, default=1)
        even_int: int = validated_field([is_positive, is_even], default=2)
        custom: int = validated_field(
            validator=lambda x: None if x == 42 else ValueError("Must be 42"),
            default=42
        )
    
    # Default values should pass validation
    n = Numbers()
    assert n.positive_int == 1
    assert n.even_int == 2
    assert n.custom == 42
    
    # Valid assignment
    n.positive_int = 10
    n.even_int = 4
    n.custom = 42
    
    # Invalid assignments
    try:
        n.positive_int = -5
        assert False, "Should have raised StrictDataclassFieldValidationError"
    except StrictDataclassFieldValidationError as e:
        assert e.field == "positive_int"
        assert "must be positive" in str(e.cause)
    
    try:
        n.even_int = 3
        assert False, "Should have raised StrictDataclassFieldValidationError"
    except StrictDataclassFieldValidationError as e:
        assert e.field == "even_int"
    
    try:
        n.custom = 99
        assert False, "Should have raised StrictDataclassFieldValidationError"
    except StrictDataclassFieldValidationError as e:
        assert e.field == "custom"

    at = AnyType(value="string")
    at.value = 123
    at.value = [1, 2, 3]
    at.value = None
    
    # Test with default_factory
    @strict

        items: List[str] = field(default_factory=list)

    wf = WithFactory()
    assert wf.items == []
    assert wf.counter == 0

    assert wf.items == ["test"]
    
    # Test that validate method cannot be overridden
    try:
        @strict
        @dataclass
        class Conflict:
            x: int
            
            def validate(self):
                pass
        
        Conflict(x=1)
        assert False, "Should have raised StrictDataclassDefinitionError"
    except StrictDataclassDefinitionError as e:
        assert "already implements a method called 'validate'" in str(e)

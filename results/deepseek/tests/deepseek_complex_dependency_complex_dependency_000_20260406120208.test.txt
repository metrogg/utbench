class TestHasBodyHeaders:
    def test_has_body_headers_with_content_length(self):
        request = Request(
            method=b"GET",
            url="http://example.com/",
            headers=[(b"Content-Length", b"100")],
            content=b"",
            extensions={}
        )
        assert has_body_headers(request) == True

    def test_has_body_headers_with_transfer_encoding(self):
        request = Request(
            method=b"GET",
            url="http://example.com/",
            headers=[(b"Transfer-Encoding", b"chunked")],
            content=b"",
            extensions={}
        )
        assert has_body_headers(request) == True

    def test_has_body_headers_without_body_headers(self):
        request = Request(
            method=b"GET",
            url="http://example.com/",
            headers=[(b"Host", b"example.com")],
            content=b"",
            extensions={}
        )
        assert has_body_headers(request) == False

    def test_has_body_headers_case_insensitive(self):
        request = Request(
            method=b"GET",
            url="http://example.com/",
            headers=[(b"content-length", b"100")],
            content=b"",
            extensions={}
        )
        assert has_body_headers(request) == True

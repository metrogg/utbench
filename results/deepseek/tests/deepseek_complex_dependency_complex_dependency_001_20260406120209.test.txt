        """Test valid paths."""
        assert _validate_path_in_repo("data/file.txt") == "data/file.txt"
        assert _validate_path_in_repo("/data/file.txt") == "data/file.txt"
        assert _validate_path_in_repo("./data/file.txt") == "data/file.txt"
        assert _validate_path_in_repo("data/subdir/file.txt") == "data/subdir/file.txt"
    
    def test_invalid_paths(self):
        """Test invalid paths."""
        with pytest.raises(ValueError, match="Invalid.*path_in_repo.*'..'"):
            _validate_path_in_repo("..")
        
        with pytest.raises(ValueError, match="Invalid.*path_in_repo.*'../data'"):
            _validate_path_in_repo("../data")
        
        with pytest.raises(ValueError, match="Invalid.*path_in_repo.*'./../data'"):
            _validate_path_in_repo("./../data")
    
    def test_forbidden_folders(self):
        """Test paths containing forbidden folders."""
        for forbidden in FORBIDDEN_FOLDERS:
            with pytest.raises(ValueError, match=f"cannot update files under a '{forbidden}/' folder"):

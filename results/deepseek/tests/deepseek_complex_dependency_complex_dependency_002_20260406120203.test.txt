    def test_count_path_segments(self):
        """Test the helper function count_path_segments."""
        # Normal cases
        assert count_path_segments("a/b/c") == 2
        assert count_path_segments("a\\b\\c") == 2  # Windows path
        assert count_path_segments("") == 0
        assert count_path_segments("file.txt") == 0
        
        # Edge cases
        assert count_path_segments("/") == 1
        assert count_path_segments("//") == 2
        assert count_path_segments("a/") == 1

    def test_split_files_and_metadata_and_archives(self, builder):
        """Test file classification into files, metadata, and archives."""
        data_files = [
            "data/file.txt",        # Regular file
            "data/file.csv",        # Regular file (second extension)
            "metadata.csv",         # Metadata file
            "archive.zip",          # Archive
            "unknown.xyz"           # Unknown file (should be ignored)
        ]
        
        files, metadata_files, archives = builder._split_files_and_metadata_and_archives(data_files)
        
        assert "data/file.txt" in files
        assert "data/file.csv" in files
        assert "metadata.csv" in metadata_files
        assert "archive.zip" in archives
        assert len(files) == 2
        assert len(metadata_files) == 1
        assert len(archives) == 1

    def test_read_metadata_csv(self, builder):
        """Test reading metadata from CSV files."""
        with tempfile.NamedTemporaryFile(mode='w', suffix='.csv', delete=False) as f:
            df = pd.DataFrame({"file_name": ["a.txt", "b.txt"], "value": [1, 2]})
            df.to_csv(f.name, index=False)
            csv_file = f.name
        
        try:
            builder.config = Mock(features=None, filters=None)
            
            tables = list(builder._read_metadata(csv_file, '.csv'))
            
            assert len(tables) > 0
            assert isinstance(tables[0], pa.Table)
            assert tables[0].num_rows == 2
        finally:
            os.unlink(csv_file)

    def test_nested_apply(self):
        """Test the _nested_apply helper function."""
        def add_one(item, path):
            if isinstance(item, int):
                return item + 1
            return item
        
        # Test with nested dictionary
        item = {"a": 1, "b": {"c": 2}}
        result = _nested_apply(item, ["b", "c"], add_one)
        assert result["b"]["c"] == 3
        
        # Test with list
        item = [1, 2, 3]
        result = _nested_apply(item, [0], add_one)
        assert result[0] == 2
        
        # Test empty path
        item = 5
        result = _nested_apply(item, [], add_one)
        assert result == 6

    def test_edge_case_all_metadata_extensions(self, builder):
        """Test with all supported metadata extensions."""
        extensions = ['.csv', '.jsonl', '.parquet']
        
        for ext in extensions:
            with tempfile.NamedTemporaryFile(suffix=ext, delete=False) as f:
                if ext == '.csv':
                    df = pd.DataFrame({"file_name": ["test.txt"]})
                    df.to_csv(f.name, index=False)
                elif ext == '.jsonl':
                    with open(f.name, 'w') as jf:
                        jf.write('{"file_name": "test.txt"}\n')
                else:  # .parquet
                    table = pa.table({"file_name": ["test.txt"]})
                    pq.write_table(table, f.name)
                
                try:
                    builder.config = Mock(features=None, filters=None)
                    tables = list(builder._read_metadata(f.name, ext))
                    assert len(tables) > 0
                    assert isinstance(tables[0], pa.Table)
                finally:
                    os.unlink(f.name)

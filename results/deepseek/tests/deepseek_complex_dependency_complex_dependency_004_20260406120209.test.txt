class TestMagicNumberBaseExtractor:
    def test_read_magic_number_success(self, tmp_path):
        """测试成功读取魔数"""
        test_file = tmp_path / "test.bin"
        test_file.write_bytes(b"\x1f\x8b\x08")
        result = MagicNumberBaseExtractor.read_magic_number(test_file, 3)
        assert result == b"\x1f\x8b\x08"

    def test_read_magic_number_file_not_found(self):
        """测试读取不存在的文件"""
        with pytest.raises(OSError):
            MagicNumberBaseExtractor.read_magic_number("/nonexistent/file", 2)

    def test_is_extractable_with_provided_magic_number(self):
        """测试使用提供的魔数判断可提取性"""
        class TestExtractor(MagicNumberBaseExtractor):
            magic_numbers = [b"\x1f\x8b"]

        # 使用正确的魔数
        assert TestExtractor.is_extractable("dummy_path", magic_number=b"\x1f\x8b")
        # 使用错误的魔数
        assert not TestExtractor.is_extractable("dummy_path", magic_number=b"\x00\x00")
        # 使用空魔数（应该从文件读取）
        with patch.object(TestExtractor, 'read_magic_number', return_value=b"\x1f\x8b"):
            assert TestExtractor.is_extractable("dummy_path", magic_number=b"")

    def test_is_extractable_file_read_error(self):
        """测试文件读取错误时的处理"""
        class TestExtractor(MagicNumberBaseExtractor):
            magic_numbers = [b"\x1f\x8b"]

        with patch.object(TestExtractor, 'read_magic_number', side_effect=OSError):
            assert not TestExtractor.is_extractable("dummy_path", magic_number=b"")

        """测试非空目录时不提取"""

        (test_dir / "file.txt").write_text("content")
        manager = ExtractManager()
        assert not manager._do_extract(test_dir, force_extract=False)

    def test_extract_unknown_format(self, tmp_path):
        """测试提取未知格式文件"""
        test_file = tmp_path / "unknown.bin"
        test_file.write_bytes(b"unknown format")
        
        manager = ExtractManager()
        with patch.object(Extractor, 'infer_extractor_format', return_value=None):

        """测试已提取的文件不再提取"""

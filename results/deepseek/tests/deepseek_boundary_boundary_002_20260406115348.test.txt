import unittest
from unittest.mock import patch, MagicMock
import sys
import os
sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

from idna import (
    IDNAError, IDNABidiError, InvalidCodepoint, InvalidCodepointContext,
    valid_label_length, valid_string_length, check_bidi, check_initial_combiner,
    check_hyphen_ok, check_nfc, valid_contextj, valid_contexto, check_label,
    alabel, ulabel, uts46_remap, encode, decode
)

class TestIDNAFunctions(unittest.TestCase):
    
    def test_valid_label_length(self):
        # Normal cases
        self.assertTrue(valid_label_length("a" * 63))
        self.assertFalse(valid_label_length("a" * 64))
        self.assertTrue(valid_label_length(b"a" * 63))
        self.assertFalse(valid_label_length(b"a" * 64))
        self.assertTrue(valid_label_length(""))
        
    def test_valid_string_length(self):
        # Without trailing dot
        self.assertTrue(valid_string_length("a" * 253, False))
        self.assertFalse(valid_string_length("a" * 254, False))
        # With trailing dot
        self.assertTrue(valid_string_length("a" * 254, True))
        self.assertFalse(valid_string_length("a" * 255, True))
        # Bytes input
        self.assertTrue(valid_string_length(b"a" * 253, False))
        self.assertFalse(valid_string_length(b"a" * 254, False))
        
    def test_check_bidi_normal(self):
        # LTR label
        self.assertTrue(check_bidi("abc"))
        # RTL label
        self.assertTrue(check_bidi("\u05d0\u05d1\u05d2"))  # Hebrew letters
        # Mixed with numbers
        self.assertTrue(check_bidi("\u05d0123"))
        
    def test_check_bidi_errors(self):
        # Unknown directionality
        with self.assertRaises(IDNABidiError):
            check_bidi("\u0000")  # NULL character
        
        # First codepoint not L/R/AL
        with self.assertRaises(IDNABidiError):
            check_bidi("1abc")  # EN as first character
        
        # Invalid direction in RTL label
        with self.assertRaises(IDNABidiError):
            check_bidi("\u05d0\u05d1\u05d2L")  # L in RTL
        
        # Invalid direction in LTR label
        with self.assertRaises(IDNABidiError):
            check_bidi("abc\u05d0")  # R in LTR
        
        # Mix numeral types in RTL
        with self.assertRaises(IDNABidiError):
            check_bidi("\u05d0\u0661\u06f1")  # AN and EN mixed
        
        # Label ends with illegal directionality
        with self.assertRaises(IDNABidiError):
            check_bidi("abc\u200c")  # BN at end
        
    def test_check_initial_combiner(self):
        self.assertTrue(check_initial_combiner("abc"))
        # Combining character at start
        with self.assertRaises(IDNAError):
            check_initial_combiner("\u0300abc")  # Combining grave
        
    def test_check_hyphen_ok(self):
        self.assertTrue(check_hyphen_ok("abc"))
        self.assertTrue(check_hyphen_ok("ab-cd"))
        
        # Hyphen at start
        with self.assertRaises(IDNAError):
            check_hyphen_ok("-abc")
        
        # Hyphen at end
        with self.assertRaises(IDNAError):
            check_hyphen_ok("abc-")
        
        # Disallowed hyphens at 3rd and 4th position
        with self.assertRaises(IDNAError):
            check_hyphen_ok("ab--cd")
            
    def test_check_nfc(self):
        # Already NFC
        check_nfc("café")
        # Not NFC
        with self.assertRaises(IDNAError):
            check_nfc("cafe\u0301")  # Decomposed
        
    def test_valid_contextj(self):
        # Test ZWJ (U+200D) with valid context
        label = "\u0d1f\u0d4d\u200d\u0d1f"  # Malayalam sequence
        self.assertTrue(valid_contextj(label, 2))
        
        # Test ZWNJ (U+200C) with invalid context
        label = "a\u200cb"
        self.assertFalse(valid_contextj(label, 1))
        
        # Test invalid positions
        label = "\u200cabc"
        self.assertFalse(valid_contextj(label, 0))
        
    def test_valid_contexto(self):
        # Middle dot (U+00B7) in valid context
        label = "l·l"
        self.assertTrue(valid_contexto(label, 1))
        
        # Middle dot in invalid context
        label = "l·m"
        self.assertFalse(valid_contexto(label, 1))
        
        # Greek ano teleia (U+0387)
        label = "α·β"
        self.assertTrue(valid_contexto(label, 1))
        
        # Hebrew punctuation
        label = "\u05d0\u05f3"  # Hebrew Geresh
        self.assertTrue(valid_contexto(label, 1))
        
        # Arabic-Indic digits
        label = "\u0661\u0662"
        self.assertTrue(valid_contexto(label, 0))
        
        # Extended Arabic-Indic digits
        label = "\u06f1\u06f2"
        self.assertTrue(valid_contexto(label, 0))
        
        # Mixed digits should fail
        label = "\u0661\u06f2"
        self.assertFalse(valid_contexto(label, 0))
        
    def test_check_label_normal(self):
        # Valid ASCII label
        check_label("example")
        # Valid Unicode label
        check_label("café")
        # Valid label with bytes input
        check_label(b"example")
        
    def test_check_label_errors(self):
        # Empty label
        with self.assertRaises(IDNAError):
            check_label("")
        
        # Invalid codepoint
        with self.assertRaises(InvalidCodepoint):
            check_label("\u0000")  # NULL
        
        # Invalid context for joiner
        with self.assertRaises(InvalidCodepointContext):
            check_label("a\u200cb")  # ZWNJ without proper context
        
        # Invalid context for CONTEXTO
        with self.assertRaises(InvalidCodepointContext):
            check_label("l·m")  # Middle dot in wrong context
        
    def test_alabel_ascii(self):
        # Pure ASCII
        self.assertEqual(alabel("example"), b"example")
        # ASCII with hyphen
        self.assertEqual(alabel("ex-ample"), b"ex-ample")
        
    def test_alabel_unicode(self):
        # Unicode to punycode
        result = alabel("café")
        self.assertTrue(result.startswith(b"xn--"))
        # Label too long
        with self.assertRaises(IDNAError):
            alabel("a" * 64)
        
    def test_ulabel_ascii(self):
        # Pure ASCII
        self.assertEqual(ulabel("example"), "example")
        # ASCII bytes
        self.assertEqual(ulabel(b"example"), "example")
        
    def test_ulabel_punycode(self):
        # A-label to Unicode
        label = alabel("café")
        self.assertEqual(ulabel(label), "café")
        # Invalid A-label
        with self.assertRaises(IDNAError):
            ulabel(b"xn--")
        # A-label ending with hyphen
        with self.assertRaises(IDNAError):
            ulabel(b"xn--test-")
            
    def test_uts46_remap(self):
        # No remapping needed
        self.assertEqual(uts46_remap("example"), "example")
        # Uppercase to lowercase
        self.assertEqual(uts46_remap("EXAMPLE"), "example")
        # Invalid codepoint
        with self.assertRaises(InvalidCodepoint):
            uts46_remap("\u0000")
            
    def test_encode_normal(self):
        # ASCII domain
        self.assertEqual(encode("example.com"), b"example.com")
        # Unicode domain
        result = encode("café.com")
        self.assertTrue(b"xn--" in result)
        # With trailing dot
        self.assertEqual(encode("example.com."), b"example.com.")
        
    def test_encode_errors(self):
        # Empty domain
        with self.assertRaises(IDNAError):
            encode("")
        # Empty label
        with self.assertRaises(IDNAError):
            encode("example..com")
        # Domain too long
        with self.assertRaises(IDNAError):
            encode("a" * 254 + ".com")
        # Invalid byte string input
        with self.assertRaises(IDNAError):
            encode(b"\xff\xfe")
            
    def test_encode_with_options(self):
        # With UTS46
        result = encode("EXAMPLE.com", uts46=True)
        self.assertEqual(result, b"example.com")
        # With std3_rules
        result = encode("example.com", std3_rules=True)
        self.assertEqual(result, b"example.com")
        # With transitional
        result = encode("example.com", transitional=True)
        self.assertEqual(result, b"example.com")
        
    def test_decode_normal(self):
        # ASCII domain
        self.assertEqual(decode("example.com"), "example.com")
        # A-label domain
        puny = alabel("café")
        self.assertEqual(decode(puny + b".com"), "café.com")
        # With trailing dot
        self.assertEqual(decode("example.com."), "example.com.")
        
    def test_decode_errors(self):
        # Empty domain
        with self.assertRaises(IDNAError):
            decode("")
        # Invalid ASCII
        with self.assertRaises(IDNAError):
            decode(b"\xff\xfe")
        # Invalid A-label
        with self.assertRaises(IDNAError):
            decode("xn--")
        # Empty label
        with self.assertRaises(IDNAError):
            decode("example..com")
            
    def test_decode_with_options(self):
        # With UTS46
        self.assertEqual(decode("EXAMPLE.com", uts46=True), "example.com")
        # With std3_rules
        self.assertEqual(decode("example.com", std3_rules=True), "example.com")
        
    def test_edge_cases(self):
        # Single character label
        check_label("a")
        self.assertEqual(alabel("a"), b"a")
        self.assertEqual(ulabel("a"), "a")
        
        # Maximum length label
        long_label = "a" * 63
        self.assertTrue(valid_label_length(long_label))
        check_label(long_label)
        
        # Label with only hyphens (invalid)
        with self.assertRaises(IDNAError):
            check_label("---")
            
        # Label starting with combining mark
        with self.assertRaises(IDNAError):
            check_label("\u0300a")
            
        # Mixed script label
        check_label("café-咖啡")  # Latin + Han
        
    def test_bidi_check_ltr(self):
        # LTR check enabled
        self.assertTrue(check_bidi("abc", check_ltr=True))
        # RTL label with LTR check
        self.assertTrue(check_bidi("\u05d0\u05d1\u05d2", check_ltr=True))
        
    def test_bytes_bytearray_input(self):
        # Bytearray input
        self.assertEqual(ulabel(bytearray(b"example")), "example")
        check_label(bytearray(b"example"))
        
        # Bytes input with non-ASCII
        with self.assertRaises(IDNAError):
            ulabel(b"caf\xc3\xa9")
            
    def test_special_characters(self):
        # Labels with dots
        self.assertEqual(encode("example。com"), b"example.com")
        self.assertEqual(decode("example。com", strict=False), "example.com")
        
        # IDNA errors propagate correctly
        with self.assertRaises(IDNAError):
            check_label("\uFFFD")  # Replacement character
            
    def test_combining_class_edge(self):
        # Test _combining_class with unknown character
        # This is an internal function but we can test through valid_contextj
        # Mock to test ValueError case
        with patch('idna._combining_class', side_effect=ValueError('Unknown character')):
            with self.assertRaises(IDNAError):
                check_label("a\u200cb")

if __name__ == '__main__':
    unittest.main()

import pytest
from datetime import datetime, date, time, timedelta
from dateutil import tz

# Test isoparser initialization
def test_isoparser_init_valid():
    parser = isoparser()
    assert parser._sep is None
    
    parser = isoparser('-')
    assert parser._sep == b'-'
    
    parser = isoparser('T')
    assert parser._sep == b'T'

def test_isoparser_init_invalid():
    with pytest.raises(ValueError):
        isoparser('12')
    
    with pytest.raises(ValueError):
        isoparser('')
    
    with pytest.raises(ValueError):
        isoparser('AB')
    
    with pytest.raises(ValueError):
        isoparser('9')

# Test isoparse function (main entry point)
def test_isoparse_basic():
    # Basic date
    assert isoparse('2023-01-15') == datetime(2023, 1, 15)
    
    # Date with time
    assert isoparse('2023-01-15T14:30:45') == datetime(2023, 1, 15, 14, 30, 45)
    
    # Date with time and microseconds
    assert isoparse('2023-01-15T14:30:45.123456') == datetime(2023, 1, 15, 14, 30, 45, 123456)
    
    # Date with time and timezone
    assert isoparse('2023-01-15T14:30:45Z') == datetime(2023, 1, 15, 14, 30, 45, tzinfo=tz.UTC)

def test_isoparse_edge_cases():
    # 24:00 handling
    assert isoparse('2023-01-15T24:00:00') == datetime(2023, 1, 16, 0, 0, 0)
    
    # Leap year
    assert isoparse('2024-02-29') == datetime(2024, 2, 29)
    
    # Minimum date
    assert isoparse('0001-01-01') == datetime(1, 1, 1)
    
    # Maximum reasonable date
    assert isoparse('9999-12-31') == datetime(9999, 12, 31)

def test_isoparse_with_custom_separator():
    parser = isoparser('T')
    assert parser.isoparse('2023-01-15T14:30:45') == datetime(2023, 1, 15, 14, 30, 45)
    
    parser = isoparser(' ')
    assert parser.isoparse('2023-01-15 14:30:45') == datetime(2023, 1, 15, 14, 30, 45)

# Test parse_isodate
def test_parse_isodate():
    parser = isoparser()
    
    # Basic dates
    assert parser.parse_isodate('2023-01-15') == date(2023, 1, 15)
    assert parser.parse_isodate('20230115') == date(2023, 1, 15)
    
    # Year-month only
    assert parser.parse_isodate('2023-01') == date(2023, 1, 1)
    
    # Year only
    assert parser.parse_isodate('2023') == date(2023, 1, 1)
    
    # Week dates
    assert parser.parse_isodate('2023-W01') == date(2023, 1, 2)  # Monday of week 1
    assert parser.parse_isodate('2023-W01-1') == date(2023, 1, 2)
    assert parser.parse_isodate('2023W011') == date(2023, 1, 2)
    
    # Ordinal dates
    assert parser.parse_isodate('2023-001') == date(2023, 1, 1)
    assert parser.parse_isodate('2023-365') == date(2023, 12, 31)
    assert parser.parse_isodate('2023365') == date(2023, 12, 31)
    
    # Leap year ordinal
    assert parser.parse_isodate('2024-366') == date(2024, 12, 31)

def test_parse_isodate_invalid():

# Test parse_isotime
def test_parse_isotime():
    parser = isoparser()
    
    # Basic times
    assert parser.parse_isotime('14:30:45') == time(14, 30, 45)
    assert parser.parse_isotime('14:30') == time(14, 30, 0)
    assert parser.parse_isotime('14') == time(14, 0, 0)
    
    # With microseconds
    assert parser.parse_isotime('14:30:45.123456') == time(14, 30, 45, 123456)
    assert parser.parse_isotime('14:30:45.123') == time(14, 30, 45, 123000)
    
    # 24:00 handling
    assert parser.parse_isotime('24:00:00') == time(0, 0, 0)
    
    # With timezone
    assert parser.parse_isotime('14:30:45Z') == time(14, 30, 45, tzinfo=tz.UTC)
    assert parser.parse_isotime('14:30:45+05:30') == time(14, 30, 45, tzinfo=tz.tzoffset(None, 19800))

def test_parse_isotime_invalid():
    parser = isoparser()
    
    with pytest.raises(ValueError):
        parser.parse_isotime('')
    
    with pytest.raises(ValueError):

    with pytest.raises(ValueError):

    with pytest.raises(ValueError):

# Test parse_tzstr
def test_parse_tzstr():
    parser = isoparser()
    
    # UTC
    assert parser.parse_tzstr('Z') == tz.UTC
    assert parser.parse_tzstr('z') == tz.UTC
    
    # Positive offsets
    assert parser.parse_tzstr('+05:30') == tz.tzoffset(None, 19800)
    assert parser.parse_tzstr('+0530') == tz.tzoffset(None, 19800)
    assert parser.parse_tzstr('+05') == tz.tzoffset(None, 18000)
    
    # Negative offsets
    assert parser.parse_tzstr('-08:00') == tz.tzoffset(None, -28800)
    assert parser.parse_tzstr('-0800') == tz.tzoffset(None, -28800)
    assert parser.parse_tzstr('-08') == tz.tzoffset(None, -28800)
    
    # Zero offset (as UTC)
    assert parser.parse_tzstr('+00:00') == tz.UTC
    assert parser.parse_tzstr('-00:00') == tz.UTC
    assert parser.parse_tzstr('+00') == tz.UTC
    
    # Zero offset (not as UTC)
    assert parser.parse_tzstr('+00:00', zero_as_utc=False) == tz.tzoffset(None, 0)

def test_parse_tzstr_invalid():
    parser = isoparser()
    
    with pytest.raises(ValueError):
        parser.parse_tzstr('')
    
    with pytest.raises(ValueError):
        parser.parse_tzstr('A')
    
    with pytest.raises(ValueError):
        parser.parse_tzstr('+25:00')
    
    with pytest.raises(ValueError):
        parser.parse_tzstr('+05:60')
    
    with pytest.raises(ValueError):
        parser.parse_tzstr('05:30')  # Missing sign
    
    with pytest.raises(ValueError):
        parser.parse_tzstr('+05:30:00')  # Too long

# Test Unicode/ASCII handling
def test_unicode_handling():
    parser = isoparser()
    
    # Unicode strings should work
    assert parser.isoparse('2023-01-15') == datetime(2023, 1, 15)
    
    # Non-ASCII should fail
    with pytest.raises(ValueError):
        parser.isoparse('2023-01-15 café')
    
    with pytest.raises(ValueError):
        parser.isoparse('café')

# Test file-like object handling
def test_file_like_object():
    class StringIO:
        def __init__(self, s):
            self.s = s
        
        def read(self):
            return self.s
    
    parser = isoparser()
    
    # Test with file-like object
    file_like = StringIO('2023-01-15T14:30:45')
    assert parser.isoparse(file_like) == datetime(2023, 1, 15, 14, 30, 45)

# Test _calculate_weekdate
def test_calculate_weekdate():
    parser = isoparser()
    
    # Week 1 of 2023
    assert parser._calculate_weekdate(2023, 1, 1) == date(2023, 1, 2)
    
    # Week 52 of 2023
    assert parser._calculate_weekdate(2023, 52, 7) == date(2023, 12, 31)
    
    # Invalid week
    with pytest.raises(ValueError):
        parser._calculate_weekdate(2023, 0, 1)
    
    with pytest.raises(ValueError):
        parser._calculate_weekdate(2023, 54, 1)
    
    # Invalid day
    with pytest.raises(ValueError):
        parser._calculate_weekdate(2023, 1, 0)
    
    with pytest.raises(ValueError):
        parser._calculate_weekdate(2023, 1, 8)

# Test mixed separators
def test_mixed_separators():
    parser = isoparser()
    
    # Should handle mixed date/time separators
    assert parser.isoparse('2023-01-15T14:30:45') == datetime(2023, 1, 15, 14, 30, 45)
    
    # Should fail with inconsistent separators in date
    with pytest.raises(ValueError):
        parser.parse_isodate('2023-W01-1')  # Mixed separator usage
    
    # Should fail with inconsistent separators in time
    with pytest.raises(ValueError):
        parser.parse_isotime('14:30')

# Test fractional seconds edge cases
def test_fractional_seconds():
    parser = isoparser()
    
    # Various fractional second formats
    assert parser.parse_isotime('14:30:45.1') == time(14, 30, 45, 100000)
    assert parser.parse_isotime('14:30:45.12') == time(14, 30, 45, 120000)
    assert parser.parse_isotime('14:30:45.123') == time(14, 30, 45, 123000)
    assert parser.parse_isotime('14:30:45.1234') == time(14, 30, 45, 123400)
    assert parser.parse_isotime('14:30:45.123456') == time(14, 30, 45, 123456)
    assert parser.parse_isotime('14:30:45.123456789') == time(14, 30, 45, 123456)  # Truncated
    
    # Comma as decimal separator
    assert parser.parse_isotime('14:30:45,123') == time(14, 30, 45, 123000)

# Test complete datetime with timezone
def test_complete_datetime_with_timezone():
    parser = isoparser()
    
    # Full datetime with timezone
    dt = parser.isoparse('2023-01-15T14:30:45+05:30')
    assert dt == datetime(2023, 1, 15, 14, 30, 45, tzinfo=tz.tzoffset(None, 19800))
    
    # With Z timezone
    dt = parser.isoparse('2023-01-15T14:30:45Z')
    assert dt == datetime(2023, 1, 15, 14, 30, 45, tzinfo=tz.UTC)
    
    # With negative timezone
    dt = parser.isoparse('2023-01-15T14:30:45-08:00')
    assert dt == datetime(2023, 1, 15, 14, 30, 45, tzinfo=tz.tzoffset(None, -28800))

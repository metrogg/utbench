class TestHasBodyHeaders:
    def test_has_body_headers_with_content_length(self):
        headers = [(b'Content-Length', b'100'), (b'Host', b'example.com')]
        request = Request(method=b'GET', url='http://example.com', headers=headers)
        assert has_body_headers(request) == True

    def test_has_body_headers_with_transfer_encoding(self):
        headers = [(b'Transfer-Encoding', b'chunked'), (b'Host', b'example.com')]
        request = Request(method=b'GET', url='http://example.com', headers=headers)
        assert has_body_headers(request) == True

    def test_has_body_headers_without_body_headers(self):
        headers = [(b'Host', b'example.com'), (b'Accept', b'*/*')]
        request = Request(method=b'GET', url='http://example.com', headers=headers)
        assert has_body_headers(request) == False

    def test_has_body_headers_case_insensitive(self):
        headers = [(b'content-length', b'100'), (b'HOST', b'example.com')]
        request = Request(method=b'GET', url='http://example.com', headers=headers)
        assert has_body_headers(request) == True

test_suite test_isoparser

# Test the isoparser class and its methods
# Coverage: isoparse, parse_isodate, parse_isotime, parse_tzstr

# ===== Normal Path Tests =====

test test_isoparse_basic_datetime
    # Test basic full datetime string parsing
    result = isoparse("2023-05-15T14:30:45")
    assert result == datetime(2023, 5, 15, 14, 30, 45)
end

test test_isoparse_datetime_with_microseconds
    # Test datetime with fractional seconds
    result = isoparse("2023-05-15T14:30:45.123456")
    assert result == datetime(2023, 5, 15, 14, 30, 45, 123456)
end

test test_isoparse_datetime_with_timezone_z
    # Test datetime with Z timezone
    result = isoparse("2023-05-15T14:30:45Z")
    assert result == datetime(2023, 5, 15, 14, 30, 45, tzinfo=tz.UTC)
end

test test_parse_isodate_basic
    # Test basic date parsing
    result = isoparser().parse_isodate("2023-05-15")
    assert result == date(2023, 5, 15)
end

test test_parse_isotime_basic
    # Test basic time parsing
    result = isoparser().parse_isotime("14:30:45")
    assert result == time(14, 30, 45)
end

test test_parse_isotime_with_microseconds
    # Test time with microseconds
    result = isoparser().parse_isotime("14:30:45.123")
    assert result == time(14, 30, 45, 123000)
end

test test_parse_isotime_hour_24
    # Test hour 24 is converted to 00:00:00 next day
    result = isoparser().parse_isotime("24:00:00")
    assert result == time(0, 0, 0)
end

# ===== Boundary Condition Tests =====

test test_isoparse_minimum_date
    # Test minimum valid date (year 0001)
    result = isoparse("0001-01-01T00:00:00")
    assert result == datetime(1, 1, 1, 0, 0, 0)
end

test test_isoparse_leap_day
    # Test February 29 in leap year
    result = isoparse("2024-02-29T12:00:00")
    assert result == datetime(2024, 2, 29, 12, 0, 0)
end

test test_isoparse_hour_24_full_datetime
    # Test hour 24 in datetime becomes next day
    result = isoparse("2023-05-15T24:00:00")
    assert result == datetime(2023, 5, 16, 0, 0, 0)
end

test test_parse_isotime_minimum
    # Test minimum time
    result = isoparser().parse_isotime("00:00:00")
    assert result == time(0, 0, 0)
end

test test_parse_isotime_maximum
    # Test maximum time before 24
    result = isoparser().parse_isotime("23:59:59")
    assert result == time(23, 59, 59)
end

test test_parse_isodate_ordinal_day_001
    # Test ordinal day 001 (January 1)
    result = isoparser().parse_isodate("2023-001")
    assert result == date(2023, 1, 1)
end

test test_parse_isodate_ordinal_day_365
    # Test ordinal day 365 (December 31, non-leap year)
    result = isoparser().parse_isodate("2023-365")
    assert result == date(2023, 12, 31)
end

test test_parse_isodate_ordinal_day_366_leap
    # Test ordinal day 366 in leap year
    result = isoparser().parse_isodate("2024-366")
    assert result == date(2024, 12, 31)
end

test test_parse_isodate_week_date
    # Test week date format YYYY-Www
    # 2023-W01-1 should be January 2, 2023 (Monday)
    result = isoparser().parse_isodate("2023-W01-1")
    assert result == date(2023, 1, 2)
end

test test_parse_isodate_week_date_no_day
    # Test week date without day (defaults to day 1)
    result = isoparser().parse_isodate("2023-W01")
    assert result == date(2023, 1, 2)
end

# ===== Custom Separator Tests =====

test test_custom_separator_t
    # Test custom separator 'T' for datetime
    parser = isoparser(sep='T')
    result = parser.isoparse("2023-05-15T14:30:45")
    assert result == datetime(2023, 5, 15, 14, 30, 45)
end

test test_custom_separator_space
    # Test custom separator space
    parser = isoparser(sep=' ')
    result = parser.isoparse("2023-05-15 14:30:45")
    assert result == datetime(2023, 5, 15, 14, 30, 45)
end

test test_custom_separator_dot
    # Test custom separator dot
    parser = isoparser(sep='.')
    result = parser.isoparse("2023-05-15.14:30:45")
    assert result == datetime(2023, 5, 15, 14, 30, 45)
end

# ===== Timezone Tests =====

test test_parse_tzstr_z_uppercase
    # Test Z timezone uppercase
    result = isoparser().parse_tzstr("Z")
    assert result == tz.UTC
end

test test_parse_tzstr_z_lowercase
    # Test z timezone lowercase
    result = isoparser().parse_tzstr("z")
    assert result == tz.UTC
end

test test_parse_tzstr_positive_offset
    # Test positive timezone offset
    result = isoparser().parse_tzstr("+05:30")
    assert result == tz.tzoffset(None, 5 * 3600 + 30 * 60)
end

test test_parse_tzstr_negative_offset
    # Test negative timezone offset
    result = isoparser().parse_tzstr("-08:00")
    assert result == tz.tzoffset(None, -8 * 3600)
end

test test_parse_tzstr_hour_only
    # Test timezone with just hours
    result = isoparser().parse_tzstr("+05")
    assert result == tz.tzoffset(None, 5 * 3600)
end

test test_parse_tzstr_no_colon
    # Test timezone without colon
    result = isoparser().parse_tzstr("+0530")
    assert result == tz.tzoffset(None, 5 * 3600 + 30 * 60)
end

test test_isoparse_with_timezone_in_datetime
    # Test full datetime with timezone
    result = isoparse("2023-05-15T14:30:45+05:30")
    assert result.hour == 14
    assert result.minute == 30
    assert result.second == 45
    assert result.tzinfo is not None
end

# ===== Error/Exception Tests =====

test test_invalid_separator_numeric
    # Separator cannot be numeric
    expect_error ValueError
    isoparser(sep='1')
end

test test_invalid_separator_too_long
    # Separator must be single character
    expect_error ValueError
    isoparser(sep='ab')
end

test test_invalid_separator_non_ascii
    # Separator must be ASCII
    expect_error ValueError
    isoparser(sep='€')
end

    expect_error ValueError

    # Test microseconds are properly scaled
    result = isoparser().parse_isotime("14:30:45.1")
    assert result == time(14, 30, 45, 100000)
end

test test_microseconds_6_digits
    # Test 6 digit microseconds
    result = isoparser().parse_isotime("14:30:45.123456")
    assert result == time(14, 30, 45, 123456)
end

    # Test microseconds are truncated to 6 digits
    result = isoparser().parse_isotime("14:30:45.1234567")
    assert result == time(14, 30, 45, 123456)
end

        self.original_outneeds = copy.deepcopy(outneeds)
        self.original_needs = copy.deepcopy(needs)

        """Restore original global state."""

        needs.update(copy.deepcopy(self.original_needs))

        """Test append_needs with a single string."""

        append_needs('math.h')

        """Test append_needs with a list of strings."""
        append_needs(['math.h', 'string.h'])

    def test_append_needs_flag_zero(self):
        """Test append_needs with flag=0 returns dependency dict."""
        result = append_needs('string', flag=0)

        """Test append_needs with unknown need type logs error."""
        captured_stderr = StringIO()
        with patch.object(sys, 'stderr', captured_stderr):
            append_needs('unknown_need_xyz')
        error_output = captured_stderr.getvalue()
        self.assertIn('unknown need', error_output)

        """Test append_needs with cppmacros entries."""
        append_needs('F_FUNC')
        self.assertIn('F_FUNC', outneeds['cppmacros'])
    
    def test_append_needs_with_cfuncs(self):
        """Test append_needs with cfuncs entries."""
        append_needs('int_from_pyobj')
        self.assertIn('int_from_pyobj', outneeds['cfuncs'])
    
    def test_append_needs_no_duplicate(self):
        """Test that append_needs does not add duplicates."""
        append_needs('math.h')
        initial_len = len(outneeds['includes0'])
        append_needs('math.h')
        self.assertEqual(len(outneeds['includes0']), initial_len)

class TestGetNeeds(unittest.TestCase):
    """Test cases for the get_needs function."""
    
    def setUp(self):
        """Save and reset global state."""
        self.original_outneeds = copy.deepcopy(outneeds)
        self.original_needs = copy.deepcopy(needs)
    
    def tearDown(self):
        """Restore original global state."""
        outneeds.clear()
        outneeds.update(copy.deepcopy(self.original_outneeds))
        needs.clear()
        needs.update(copy.deepcopy(self.original_needs))
    
    def test_get_needs_returns_dict(self):
        """Test that get_needs returns a dictionary."""
        result = get_needs()
        self.assertIsInstance(result, dict)
    
    def test_get_needs_includes_all_keys(self):
        """Test that get_needs includes all expected keys."""
        result = get_needs()
        expected_keys = ['includes0', 'includes', 'typedefs', 'typedefs_generated',
                        'userincludes', 'cppmacros', 'cfuncs', 'callbacks',
                        'f90modhooks', 'commonhooks']
        for key in expected_keys:
            self.assertIn(key, result)
    
    def test_get_needs_empty_outneeds(self):
        """Test get_needs with empty outneeds."""
        for key in outneeds:
            outneeds[key] = []
        result = get_needs()
        self.assertEqual(result['includes0'], ['includes0'])
    
    def test_get_needs_with_simple_entries(self):
        """Test get_needs with simple entries."""
        outneeds['includes0'] = ['math.h']
        result = get_needs()
        self.assertIn('math.h', result['includes0'])
    
    def test_get_needs_respects_dependencies(self):
        """Test get_needs respects dependency ordering."""
        needs['test_dep'] = ['math.h']
        outneeds['includes0'] = ['test_dep', 'math.h']
        result = get_needs()
        math_idx = result['includes0'].index('math.h')
        if 'test_dep' in result['includes0']:
            test_idx = result['includes0'].index('test_dep')
            self.assertLess(math_idx, test_idx)

class TestIntegration(unittest.TestCase):
    """Integration tests for the module."""
    
    def setUp(self):
        """Save original state."""
        self.original_outneeds = copy.deepcopy(outneeds)
        self.original_needs = copy.deepcopy(needs)
    
    def tearDown(self):
        """Restore original state."""
        outneeds.clear()
        outneeds.update(copy.deepcopy(self.original_outneeds))
        needs.clear()
        needs.update(copy.deepcopy(self.original_needs))
    
    def test_append_and_get_needs_workflow(self):
        """Test complete workflow of append_needs followed by get_needs."""
        append_needs('math.h')
        append_needs('string.h')
        result = get_needs()
        self.assertIsInstance(result, dict)
        self.assertIn('includes0', result)
    
    def test_nested_dependencies_resolution(self):
        """Test that nested dependencies are properly resolved."""
        needs['level1'] = ['level2']
        needs['level2'] = ['math.h']
        outneeds['includes0'] = ['level1']
        result = get_needs()
        self.assertIn('level1', result.get('includes0', []))
        self.assertIn('level2', result.get('includes0', []))
